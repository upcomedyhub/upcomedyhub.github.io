1. User Schema (models/User.js)
------------------------------------------
const mongoose = require('mongoose');

const UserSchema = new mongoose.Schema({
  name: { type: String, required: true },
  email: { type: String, required: true, unique: true },
  password: { type: String, required: true },
  permanent_memory: { 
    type: [String], 
    default: ["User name is Mayank Bhai. He is the creator/director of Zyro AI."] 
  },
  created_at: { type: Date, default: Date.now }
});

module.exports = mongoose.model('User', UserSchema);

2. Chat Schema (models/Chat.js)
------------------------------------------
const mongoose = require('mongoose');

const ChatSchema = new mongoose.Schema({
  user_id: { type: mongoose.Schema.Types.ObjectId, ref: 'User', required: true },
  title: { type: String, default: 'Nayi Baatcheet 💬' },
  created_at: { type: Date, default: Date.now }
});

module.exports = mongoose.model('Chat', ChatSchema);

3. Message Schema (models/Message.js)
------------------------------------------
const mongoose = require('mongoose');

const MessageSchema = new mongoose.Schema({
  chat_id: { type: mongoose.Schema.Types.ObjectId, ref: 'Chat', required: true },
  role: { type: String, enum: ['user', 'model'], required: true },
  text: { type: String, required: true },
  has_attachment: { type: Boolean, default: false },
  attachment_url: { type: String, default: null },
  timestamp: { type: Date, default: Date.now }
});

module.exports = mongoose.model('Message', MessageSchema);
